5. CDS View Layer for RAP

Interface View ZI_

@AbapCatalog.sqlViewName: 'ZIVIEW'
@AbapCatalog.compiler.compareFilter: true
@AccessControl.authorizationCheck: #NOT_REQUIRED
@EndUserText.label: 'Z Interface View'
define view ZI_BusinessObject
  with parameters
    p_active as abap.bool
{
  key OBJ_ID      as Id,
       OBJ_NAME    as Name,
       CREATED_ON  as Created,
       CHANGED_ON  as Changed,
       STATUS      as Status
}
where p_active = true;

Projection View ZC_ with UI Annotations

@AbapCatalog.sqlViewName: 'ZCVIEW'
@AbapCatalog.compiler.compareFilter: true
@EndUserText.label: 'Z Projection View'
@UI.dataCategory: #DYNAMIC
@UI.lineItem: [
  { position: 1, label: 'Object ID',   importance: #HIGH   },
  { position: 2, label: 'Object Name', importance: #HIGH   },
  { position: 3, label: 'Created On',  importance: #MEDIUM },
  { position: 4, label: 'Status',      importance: #MEDIUM }
]
define view ZC_BusinessObject
  as select from ZI_BusinessObject
{
  key Id      as OBJ_ID,
       Name    as OBJ_NAME,
       Created as CREATED_ON,
       Status  as OBJ_STATUS
}
